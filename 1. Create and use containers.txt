# show running containers
docker container ls

# show all containers
docker container ls -a

# run with a name
docker container run --publish 80:80 --detach --name webhost nginx

# get log events of a container
docker container logs <container>

# get processes in a container
docker container top <container>
# matching processes only works on standalone Linux
ps aux

# remove container
docker container rm <container>

# remove container (including running)
docker container rm -f <container>

# get live streaming data of containers
docker container stats

# in a container
# -it t is psudo tty, i keeps the session open
## with just i, your prompt is open but there is no connect between the container and your keyboard, so you cannot ls or exit.
## with just t, you can have the prompt but there is not foramtting in the. 
# this container only runs as far as the command on the startup lines, in this case "bash"
docker container run -it --name proxy nginx bash

# start a ubuntu container (container has minimal installation of ubuntu)
# it also starts the bin/bash instead of the bash pointer
docker container run -it --name ubuntu ubuntu
# you exit and get back in (you dont need -t as the container is already built with tty)
docker container start -ai ubuntu

# open an interactive Bash shell session inside an already running Docker container
# run additional commands in an existing container, for example for troubleshooting
docker container exec -it <container> bash

# cannot run things not in the image, ex. alpine doesn't have bash
docker container run -it alpine sh

-------------network-------------------

# get the container ports
docker container port <container>

# find ip of a running container:
docker inspect <container_id>  --format='{{range.NetworkSettings.Networks}}{{.IPAddress}}{{end}}'

# get to nginx passing proxy
curl --noproxy '*' <IPAddress>

# return default brdige Network
docker network inspect brdige

# create network
docker network create my_app_net

# create container in a network
# might need to stop other containers allocated the same port
docker container run --publish 80:80 --detach --name webhost_in_network --network my_app_net nginx

# add an existing container to a network
docker network connect <network> <container_id>

# remove an existing container from a network
docker network disconnect <network> <container_id>

# ping another container using DNS 
# the default bridge network doesn't assign dns name for containers
# so it is easier to simply create a new network
docker container exec -it my_nginx ping new_nginx
