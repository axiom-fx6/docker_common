# installation
# https://docs.astral.sh/uv/getting-started/installation/

# create a simple project structure (most cases)
uv init --python 3.11 app

# install pkg. this also creates .venv
uv add flask requests

# get dependecy tree
uv tree

# In terminal, no need to activate venv

uv run main.py
